# RealmRaid 结界突破 任务详解

## 1. 文件概述

**任务功能**: 自动执行结界突破，包括：
- 切换御魂配置
- 检查突破券数量
- 按勋章优先级选择目标
- 处理呱太活动
- 刷新对手列表
- 处理战斗结果

**入口文件**: `tasks/RealmRaid/script_task.py`

---

## 2. 配置参数说明

**配置文件**: `tasks/RealmRaid/config.py`

### RaidConfig 参数

| 参数名 | 类型 | 默认值 | 说明 |
|--------|------|--------|------|
| `number_attack` | int | 30 | 挑战次数上限 (1-30) |
| `number_base` | int | 0 | 突破券基准值 (0-20) |
| `exit_four` | bool | True | 是否退出四次（退四打九策略） |
| `order_attack` | str | '5 > 4 > 3 > 2 > 1 > 0' | 勋章优先级顺序 |
| `three_refresh` | bool | False | 三次后是否刷新 |
| `when_attack_fail` | WhenAttackFail | REFRESH | 失败后处理方式 |

### WhenAttackFail 枚举

| 值 | 说明 |
|----|------|
| `EXIT` | 失败后退出 |
| `CONTINUE` | 失败后继续 |
| `REFRESH` | 失败后刷新 |

### RealmRaid 配置结构

| 参数名 | 类型 | 说明 |
|--------|------|------|
| `scheduler` | Scheduler | 调度器配置 |
| `raid_config` | RaidConfig | 突破配置 |
| `general_battle_config` | GeneralBattleConfig | 战斗配置 |
| `switch_soul_config` | SwitchSoulConfig | 御魂配置 |

## 4. 核心代码逐行解释

### ScriptTask.run() - 主入口

```python
def run(self):
    self.run_2()
```

### run_2() - 主执行逻辑

```python
def run_2(self):
    con = self.config.realm_raid

    # 切换御魂
    if con.switch_soul_config.enable:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul(con.switch_soul_config.switch_group_team)
    if con.switch_soul_config.enable_switch_by_name:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul_by_name(...)

    # 导航到结界突破
    self.ui_get_current_page()
    self.ui_goto(page_realm_raid)

    # 处理呱太活动弹窗
    self.screenshot()
    if self.appear(self.I_FROG_RAID):
        while 1:
            if not self.appear(self.I_FROG_RAID):
                break
            if self.appear_then_click(self.I_FROG_RAID, interval=1):
                continue

    # 确保锁定阵容
    self.ensure_lock(con.general_battle_config.lock_team_enable)

    # 判断是否呱太活动
    frog = self.is_frog(True)

    # 主循环
    success = True
    last_battle = True
    while 1:
        self.screenshot()
        # 处理刷新确认弹窗
        if self.appear(self.I_FRESH_ENSURE):
            self.appear_then_click(self.I_FRESH_ENSURE, interval=1.5)
            continue
        # 检查票数
        if not self.check_ticket(con.raid_config.number_base):
            break
        # 检查挑战次数
        if self.current_count >= con.raid_config.number_attack:
            break

        # 寻找目标
        medal, index = self.find_one(False)
        if not medal and not index:
            # 无可挑战目标
            if con.raid_config.when_attack_fail == WhenAttackFail.CONTINUE:
                if self.check_refresh():
                    continue
                else:
                    success = False
                    break
            else:
                success = False
                break

        # 判断是否左上角第一个（退四打九）
        if index == 1:
            if con.raid_config.exit_four:
                # 退出四次
                for _ in range(4):
                    self.fire(index)
                    self.run_general_battle_back(con.general_battle_config, exit_four=True)

        # 检查是否呱太
        if self.check_medal_is_frog(frog, medal, index):
            con.general_battle_config.lock_team_enable = False

        # 挑战
        self.fire(index)
        last_battle = self.run_general_battle(con.general_battle_config)

        # 检查三胜奖励
        if self.reward_detect_click(False):
            continue

        # 三次刷新
        if con.raid_config.three_refresh and self.appear(self.I_RR_THREE):
            if self.check_refresh():
                continue
            else:
                success = False
                break

        # 失败处理
        if not last_battle:
            if con.raid_config.when_attack_fail == WhenAttackFail.REFRESH:
                if self.check_refresh():
                    continue
                else:
                    success = False
                    break
            elif con.raid_config.when_attack_fail == WhenAttackFail.EXIT:
                break

    # 退出
    self.ui_click(self.I_BACK_RED, self.I_CHECK_EXPLORATION)
    self.ui_get_current_page()
    self.ui_goto(page_main)
    self.set_next_run(task='RealmRaid', success=success, finish=True)
    raise TaskEnd
```

### find_one() - 寻找目标

```python
def find_one(self, screenshot: bool = True) -> tuple:
    if screenshot:
        self.screenshot()
    image = self.device.image

    # 处理失败标记
    if self.config.realm_raid.raid_config.when_attack_fail == WhenAttackFail.CONTINUE:
        for i, roi in enumerate(self.false_roi):
            self.false_image.roi_back = roi
            if not self.appear(self.false_image):
                continue
            # 清除失败位置
            x, y, w, h = self.partition[i].roi_back
            image[y:y+h, x:x+w, ...] = 0

    # 按优先级查找勋章
    target = self.order_medal.find_anyone(image)
    if target:
        center = target.front_center()
        # 确定位置索引
        for i, click in enumerate(self.partition):
            x1, x2 = click.roi_front[0], click.roi_front[0] + click.roi_front[2]
            y1, y2 = click.roi_front[1], click.roi_front[1] + click.roi_front[3]
            if x1 < center[0] < x2 and y1 < center[1] < y2:
                return target, i + 1

    return None, None
```

### fire() - 挑战

```python
def fire(self, order: int):
    click = self.partition[order - 1]

    # 等待回到主界面
    while not self.appear(self.I_RR_PERSON):
        # 处理各种弹窗
        if self.appear(self.I_FRESH_ENSURE):
            self.device.click(x=530, y=460)  # 点取消
            time.sleep(1.5)
            continue
        self.device.click(x=640, y=50)  # 点击安全区域
        time.sleep(1.5)

    # 进攻循环
    while 1:
        self.screenshot()
        if self.appear(self.I_FRESH_ENSURE):
            self.device.click(x=530, y=460)
            continue
        if not self.appear(self.I_RR_PERSON, threshold=0.8):
            break
        if self.appear_then_click(self.I_FIRE, interval=1):
            continue
        if self.click(click, interval=1.8):
            continue
```

### check_ticket() - 检查票数

```python
def check_ticket(self, base: int = 0) -> bool:
    if base < 0 or base > 30:
        base = 0

    self.wait_until_appear(self.I_BACK_RED)
    self.screenshot()
    cu, res, total = self.O_NUMBER.ocr(self.device.image)

    if total == 0:
        self.reward_detect_click(True)
        cu, res, total = self.O_NUMBER.ocr(self.device.image)

    if cu == 0 and cu + res == total:
        return False  # 无票
    elif cu + res == total and cu < base:
        return False  # 票不足基准

    return True
```

## 6. 设计模式总结

### 1. 策略模式
- `order_attack` 定义勋章优先级
- `when_attack_fail` 定义失败处理策略
- `exit_four` 控制退四打九

### 2. 状态机模式
- 检测票数、挑战次数、锁定状态
- 处理各种弹窗和异常状态

### 3. 图像网格匹配
- `ImageGrid` 实现多勋章同时匹配
- `partition` 定义9宫格位置

### 4. 重试机制
- `check_ticket()` 处理票数识别异常
- `reward_detect_click()` 处理奖励弹窗

### 5. 呱太活动适配
- `is_frog()` 检测呱太活动
- `check_medal_is_frog()` 识别呱太勋章
- 动态调整锁定策略
