# FallenSun 日轮之城 任务详解

## 1. 文件概述

**任务功能**: 自动挑战日轮之城御魂副本，支持多种游戏身份模式：
- **队长模式 (LEADER)**: 创建队伍、邀请队友、循环挑战
- **队员模式 (MEMBER)**: 接受邀请、等待队长开始战斗
- **单人模式 (ALONE)**: 独自挑战副本
- **野队模式 (WILD)**: 未实现（仅记录日志）

日轮之城是阴阳师中的御魂副本之一，提供日蚀套御魂。

**文件结构**:
- `script_task.py` - 任务主逻辑入口
- `config.py` - 配置参数定义
- `assets.py` - 图像/点击/OCR等资源定义

### run() - 主入口方法

```python
def run(self) -> bool:
    # 御魂切换方式一：按组号/队号
    if self.config.fallen_sun.switch_soul.enable:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul(self.config.fallen_sun.switch_soul.switch_group_team)

    # 御魂切换方式二：按名称
    if self.config.fallen_sun.switch_soul.enable_switch_by_name:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul_by_name(...)

    # 初始化限制条件
    limit_count = self.config.fallen_sun.fallen_sun_config.limit_count
    limit_time = self.config.fallen_sun.fallen_sun_config.limit_time
    self.current_count = 0
    self.limit_count: int = limit_count
    self.limit_time: timedelta = timedelta(...)

    self.ui_get_current_page()
    self.ui_goto(page_main)
    config: FallenSun = self.config.fallen_sun

    # 根据用户身份选择执行模式
    success = True
    match config.fallen_sun_config.user_status:
        case UserStatus.LEADER: success = self.run_leader()
        case UserStatus.MEMBER: success = self.run_member()
        case UserStatus.ALONE:  self.run_alone()
        case UserStatus.WILD:   self.run_wild()      # 未实现

    # 设置下次运行时间
    if success:
        self.set_next_run('FallenSun', finish=True, success=True)
    else:
        self.set_next_run('FallenSun', finish=False, success=False)

    raise TaskEnd
```

### check_lock() - 检查阵容锁定

```python
def check_lock(self, lock: bool = True) -> bool:
    logger.info('Check lock: %s', lock)
    if lock:
        while 1:
            self.screenshot()
            if self.appear(self.I_FALLEN_SUN_LOCK): return True       # 已锁定
            if self.appear_then_click(self.I_FALLEN_SUN_UNLOCK, interval=1): continue  # 点击锁定
    else:
        while 1:
            self.screenshot()
            if self.appear(self.I_FALLEN_SUN_UNLOCK): return True     # 已解锁
            if self.appear_then_click(self.I_FALLEN_SUN_LOCK, interval=1): continue    # 点击解锁
```

### run_leader() - 队长模式

```python
def run_leader(self):
    # 导航到日轮之城
    self.ui_get_current_page()
    self.ui_goto(page_soul_zones)
    self.fallen_sun_enter()

    # 选层、锁阵容
    layer = self.config.fallen_sun.fallen_sun_config.layer
    self.check_layer(layer)
    self.check_lock(self.config.fallen_sun.general_battle_config.lock_team_enable)

    # 创建队伍和房间
    while 1:
        self.screenshot()
        if self.appear(self.I_CHECK_TEAM): break
        if self.appear_then_click(self.I_FORM_TEAM, interval=1): continue
    self.create_room()
    self.ensure_private()
    self.create_ensure()

    # 主循环：邀请队友并战斗
    success = True
    is_first = True
    while 1:
        self.screenshot()
        if self.check_and_invite(self.config.fallen_sun.invite_config.default_invite):
            continue
        if self.appear_then_click(self.I_PET_PRESENT, action=self.C_WIN_3, interval=1):
            continue
        # 检查限制条件
        if self.current_count >= self.limit_count:
            if self.is_in_room(): break
        if datetime.now() - self.start_time >= self.limit_time:
            if self.is_in_room(): break
        # 检查房间存活
        if not self.is_in_room():
            sleep(0.5)
            if self.appear(self.I_MATCHING) or self.appear(self.I_CHECK_EXPLORATION):
                sleep(0.5)
                if self.appear(self.I_MATCHING) or self.appear(self.I_CHECK_EXPLORATION):
                    success = False; break
            continue
        # 邀请并战斗
        if not is_first:
            if self.run_invite(config=...):
                self.run_general_battle(config=...)
            else:
                success = False; break
        if is_first:
            if not self.run_invite(config=..., is_first=True):
                success = False; break
            else:
                is_first = False
                self.run_general_battle(config=...)

    # 退出清理
    self.exit_room()
    self.exit_team()
    self.ui_get_current_page()
    self.ui_goto(page_main)
    return success
```

### run_member() - 队员模式

```python
def run_member(self):
    self.ui_get_current_page()
    self.device.stuck_record_add('BATTLE_STATUS_S')
    while 1:
        self.screenshot()
        if self.appear_then_click(self.I_PET_PRESENT, action=self.C_WIN_3, interval=1):
            continue
        # 检查限制
        if self.current_count >= self.limit_count: break
        if datetime.now() - self.start_time >= self.limit_time: break
        # 接受邀请
        if self.check_then_accept(): continue
        # 在房间中等待战斗
        if self.is_in_room():
            self.device.stuck_record_clear()
            if self.wait_battle(wait_time=...):
                self.run_general_battle(config=...)
            else: break
        elif self.check_take_over_battle(False, config=...):
            continue
    # 退出清理
    while 1:
        if self.appear(self.I_GI_HOME) or self.appear(self.I_GI_EXPLORE): break
        if self.exit_room(): pass
        if self.exit_battle(): pass
    self.ui_get_current_page()
    self.ui_goto(page_main)
    return True
```

### run_alone() - 单人模式

```python
def run_alone(self):
    # 导航到日轮之城
    self.ui_get_current_page()
    self.ui_goto(page_soul_zones)
    self.fallen_sun_enter()
    self.check_layer(layer)
    self.check_lock(...)

    # 检测函数
    def is_in_fallen_sun(screenshot=False) -> bool:
        if screenshot: self.screenshot()
        return self.appear(self.I_FALLEN_SUN_FIRE)

    # 主战斗循环
    while 1:
        self.screenshot()
        if not is_in_fallen_sun(): continue
        if self.current_count >= self.limit_count: break
        if datetime.now() - self.start_time >= self.limit_time: break
        # 点击挑战
        while 1:
            self.screenshot()
            if self.appear_then_click(self.I_FALLEN_SUN_FIRE, interval=1): pass
            if not self.appear(self.I_FALLEN_SUN_FIRE):
                self.run_general_battle(config=...)
                break

    # 退出返回主页
    while 1:
        if not self.appear(self.I_FORM_TEAM): break
        if self.appear_then_click(self.I_BACK_BL, interval=1): continue
    self.ui_current = page_soul_zones
    self.ui_goto(page_main)
```

### battle_wait() - 重写战斗等待

```python
def battle_wait(self, random_click_swipt_enable: bool) -> bool:
    self.device.stuck_record_add('BATTLE_STATUS_S')
    self.device.click_record_clear()
    logger.info("Start battle process")
    win: bool = False

    # 阶段1：检测战斗结果
    while 1:
        self.screenshot()
        # 检测胜利（支持封魔的特殊胜利图片）
        if self.appear(self.I_WIN, threshold=0.8) or self.appear(self.I_DE_WIN):
            if self.appear(self.I_DE_WIN):
                self.ui_click_until_disappear(self.I_DE_WIN)
            win = True; break
        # 检测失败
        if self.appear(self.I_FALSE, threshold=0.8):
            win = False; break
        # 检测奖励界面
        if self.appear(self.I_REWARD, threshold=0.6): win = True; break
        if self.appear(self.I_REWARD_GOLD, threshold=0.8): win = True; break
        if self.appear(self.I_REWARD_PURPLE_SNAKE_SKIN, threshold=0.8): win = True; break
        # 随机滑动防封
        if random_click_swipt_enable: self.random_click_swipt()

    # 阶段2：确认战斗结果
    while 1:
        self.screenshot()
        if win:
            action_click = random.choice([self.C_WIN_1, self.C_WIN_2, self.C_WIN_3])
            if self.appear_then_click(self.I_WIN, action=action_click, interval=0.5): continue
            if not self.appear(self.I_WIN): break
        else:
            if self.appear_then_click(self.I_FALSE, threshold=0.6): continue
            if not self.appear(self.I_FALSE, threshold=0.6): return False

    # 阶段3：领取奖励
    if not self.wait_until_appear(self.I_REWARD):
        return win  # 无奖励直接返回
    while 1:
        self.screenshot()
        action_click = random.choice([self.C_REWARD_1, self.C_REWARD_2, self.C_REWARD_3])
        if (self.appear_then_click(self.I_REWARD, action=action_click, interval=1.5) or
            self.appear_then_click(self.I_REWARD_GOLD, action=action_click, interval=1.5) or
            self.appear_then_click(self.I_REWARD_PURPLE_SNAKE_SKIN, action=action_click, interval=1.5)):
            continue
        if (not self.appear(self.I_REWARD) and
            not self.appear(self.I_REWARD_GOLD) and
            not self.appear(self.I_REWARD_PURPLE_SNAKE_SKIN)):
            break
    return win
```

## 6. 设计模式总结

### 1. 策略模式 (Strategy Pattern)
- `UserStatus` 枚举决定执行模式
- `Layer` 枚举决定挑战层数
- `match` 语句实现策略分发

### 2. 模板方法模式 (Template Method Pattern)
- `run()` 定义任务执行模板
- `battle_wait()` 重写父类方法以适配日轮之城的特殊奖励类型（蛇皮、金币等）

### 3. 组合继承 (Mixin Pattern)
- 通过多重继承组合6个Mixin的能力
- 与八岐大蛇共享相同的继承结构

### 4. 重试与容错
- 房间存活检测使用双重确认（sleep + 二次检测）
- `battle_wait()` 支持多种奖励类型的识别
- `run_wild()` 未实现，直接记录错误日志

### 5. 与八岐大蛇的代码复用
- 日轮之城的代码结构与八岐大蛇高度相似
- 主要区别在于：资源图像不同、层数枚举不同、`battle_wait()` 的奖励处理不同
- `run_alone()` 使用 `I_BACK_BL` 而非 `I_UI_BACK_YELLOW` 返回
- `run_member()` 使用 `I_GI_HOME` / `I_GI_EXPLORE` 而非 `I_CHECK_MAIN` / `I_CHECK_EXPLORATION` 判断退出
