# SixRealms 六道之门 任务详解

## 1. 文件概述

**任务功能**: 自动挑战六道之门副本（月海模式），这是一个Roguelike玩法的自动化实现。六道之门包含多种岛屿类型，玩家需要在岛屿间导航、战斗、购买技能，最终击败Boss。

支持的岛屿类型：
- **宁息之岛 (101)**: 商店，购买"柔风"技能
- **神秘之岛 (102)**: 技能仿造/转移
- **混乱之岛 (103)**: 宝箱或精英战斗
- **战斗之岛 (104)**: 精英怪物战斗
- **星之岛 (105)**: 普通小怪战斗
- **Boss岛 (106)**: 最终Boss月读

**文件结构**:
- `script_task.py` - 任务主入口
- `config.py` - 配置参数定义
- `assets.py` - 资源定义（大量图像/点击/OCR/Swipe资源）
- `common.py` - 通用枚举定义
- `oas_ocr.py` - 自定义OCR（竖排文字识别）
- `moon_sea/` - 月海子模块目录
  - `moon_sea.py` - 月海主逻辑
  - `map.py` - 地图OCR识别与岛屿导航
  - `skills.py` - 技能选择与战斗基础
  - `l101.py` - 宁息之岛逻辑
  - `l102.py` - 神秘之岛逻辑
  - `l103.py` - 混乱之岛逻辑
  - `l104.py` - 战斗之岛逻辑
  - `l105.py` - 星之岛逻辑

```python
class ScriptTask(GameUi, SwitchSoul, MoonSea):

    @property
    def _config(self):
        return self.config.model.six_realms    # 配置访问属性

    def run(self):
        # 切换御魂
        if self._config.switch_soul_config.enable:
            self.ui_get_current_page()
            self.ui_goto(page_shikigami_records)
            self.run_switch_soul(self._config.switch_soul_config.switch_group_team)
        if self._config.switch_soul_config.enable_switch_by_name:
            self.ui_get_current_page()
            self.ui_goto(page_shikigami_records)
            self.run_switch_soul_by_name(...)

        # 导航到六道之门
        self.ui_get_current_page()
        self.ui_goto(page_six_gates)

        # 执行月海
        self.run_moon_sea()

        # 退出六道
        while 1:
            self.screenshot()
            if self.appear(self.I_CHECK_EXPLORATION) or self.appear(self.I_CHECK_MAIN):
                break
            if self.appear_then_click(self.I_BACK_EXIT, interval=2):
                continue

        self.set_next_run('SixRealms', success=True, finish=True)
        raise TaskEnd

    def run_moon_sea(self):
        self._run_moon_sea()                   # 调用MoonSea的主逻辑
        self.ui_click(self.I_BACK_EXIT, self.I_CHECK_SIX_GATES)  # 退出月海
```

```python
class VerticalText(BaseCor):
    """竖排文字OCR，处理六道之门中的竖排文字"""

    @staticmethod
    def rotate_image(image):
        """将竖排文字图片旋转90度变为横排"""
        height, width = image.shape[0:2]
        if height * 1.0 / width >= 1.5:   # 高宽比>=1.5认为是竖排
            image = np.rot90(image)
        return image

    def detect_and_ocr(self, *args, **kwargs):
        """重写检测+OCR方法，支持竖排文字"""

class StoneOcr(VerticalText):
    """石头文字OCR，用于识别地图上的岛屿名称"""

    def pre_process(self, image):
        """图像预处理：增强石头文字的对比度"""
        # YUV色彩空间提取U通道（增强特定颜色）
        yuv = cv2.cvtColor(image, cv2.COLOR_RGB2YUV)
        _, u, _ = cv2.split(yuv)
        cv2.subtract(128, u, dst=u)
        cv2.multiply(u, 8, dst=u)

        # 颜色相似度（目标色：(234, 213, 181) 石头色）
        color = color_similarity_2d(image, color=(234, 213, 181))

        # HSV提取V通道（亮度）
        hsv = cv2.cvtColor(image, cv2.COLOR_RGB2HSV)
        _, _, v = cv2.split(hsv)

        # 应用多层掩码增强文字
        image = apply_mask(image, u)
        image = apply_mask(image, color)
        image = apply_mask(image, v)

        # 增强对比度并反转
        cv2.convertScaleAbs(image, alpha=3, dst=image)
        cv2.subtract((255, 255, 255, 0), image, dst=image)
        return image
```

```python
class MoonSeaSkills(BaseTask, SixRealmsAssets):

    cnt_skill101 = 0   # 柔风技能等级计数

    def in_main(self, screenshot=False) -> bool:
        """判断是否在月海主界面"""
        if screenshot: self.screenshot()
        if self.appear(self.I_M_STORE): return True         # 商店图标
        if self.appear(self.I_M_STORE_ACTIVITY): return True # 可购买商品
        if self.appear(self.I_BOSS_FIRE): return True       # Boss挑战
        return False

    def battle_lock_team(self):
        """锁定战斗阵容"""
        self.ui_click(self.I_BATTLE_TEAM_UNLOCK, self.I_BATTLE_TEAM_LOCK)

    def island_battle(self):
        """岛屿小怪战斗"""
        while 1:
            self.screenshot()
            if self.appear(self.I_SKILL_REFRESH): break     # 技能选择界面
            if self.appear(self.I_COIN): break              # 金币奖励
            if self.appear(self.I_SELECT_SHIKIGAMI_AND_CONFIRM):
                self.select_shikigami_and_confirm()          # 选择式神
                continue
            if self.appear_then_click(self.I_NPC_FIRE, interval=1):
                self.device.stuck_record_clear()
                self.device.stuck_record_add('BATTLE_STATUS_S')
                continue

    def _select_skill(self) -> int:
        """选择技能，优先选择柔风(101)"""
        self.wait_until_stable(self.I_SELECT_3)
        select = 3  # 默认选第4个（最右边）
        button = None
        # 优先选柔风
        if self.appear(self.I_SKILL101):
            self.cnt_skill101 += 1
            button = self.I_SKILL101
        # 其次选洞察之力
        elif self.appear(self.I_SKILL105):
            button = self.I_SKILL105

        # 根据技能位置确定点击哪个选择按钮
        if button is not None:
            x, y = button.front_center()
            if x < 360: select = 0
            elif 360 <= x < 640: select = 1
            elif 640 <= x < 960: select = 2
            else: select = 3
        return select

    def select_skill(self, refresh=False):
        """战斗结束后选择技能"""
        def check_coin_skill() -> bool:
            coin = self.O_COIN_NUM.ocr(self.device.image)
            return False if coin < 50 else True

        def check_refresh() -> bool:
            text = self.O_SKILL_REFRESH.ocr(self.device.image)
            matches = re.search(r"剩\d+次", text)
            if matches:
                refresh_time = int(matches.group()[1])
                return refresh_time > 0
            return False

        while 1:
            self.screenshot()
            if self.in_main(): break

            if self.appear(self.I_UI_CONFIRM):
                self.ui_click_until_disappear(self.I_UI_CONFIRM)

            # 技能选择界面
            if self.appear(self.I_SKILL_REFRESH) and self.appear(self.I_SELECT_3) and not self.appear(self.I_COIN2):
                select = self._select_skill()
                # 如果没有柔风、钱够、有刷新次数、柔风未满级，则刷新
                if refresh and select == 3 and check_coin_skill() and check_refresh() and self.cnt_skill101 < 5:
                    self.appear_then_click(self.I_SKILL_REFRESH)
                    self.wait_until_stable(self.I_UI_CONFIRM, timeout=Timer(2))
                    self.appear_then_click(self.I_UI_CONFIRM)
                    self.wait_animate_stable(self.C_MAIN_ANIMATE_KEEP, timeout=1)
                    continue
                if self.appear_then_click(self.selects_button[select]):
                    self.wait_animate_stable(self.C_MAIN_ANIMATE_KEEP, timeout=2)
                    continue
            # 领取金币奖励
            if self.appear_then_click(self.I_COIN, action=self.C_UI_REWARD, interval=1.5):
                continue
```

```python
class MoonSeaMap(MoonSeaSkills):
    map_ocr = StoneOcr(...)    # 使用自定义石头OCR

    def decide(self) -> tuple:
        """
        通过OCR识别地图上的岛屿
        返回: (岛屿类型, 剩余回合数, ROI坐标)
        """
        self.screenshot()
        # 检测Boss
        if self.appear(self.I_BOSS_FIRE):
            return MoonSeaType.island106, 0, (0, 0, 0, 0)

        isl_type = MoonSeaType.island100
        results = self.map_ocr.detect_and_ocr(image=self.device.image)

        for result in results:
            x1, y1, x2, y2 = result.box[0][0], result.box[0][1], result.box[2][0], result.box[2][1]
            w, h = x2 - x1, y2 - y1
            text = result.ocr_text

            # 提取剩余回合数
            if self.contains_any_char(text, chars='回合国') and (x1 > 1000 and y1 + h < 300):
                match = re.search(r'\d{1,2}', text)
                if match: isl_num = int(match.group())

            # 根据文字内容判断岛屿类型
            if '宁息' in text:     isl_type = MoonSeaType.island101
            elif '神秘' in text:   isl_type = MoonSeaType.island102
            elif '回混范' in text: isl_type = MoonSeaType.island103
            elif '蜜馨屡战' in text: isl_type = MoonSeaType.island104
            elif '星之' in text:   isl_type = MoonSeaType.island105

        return isl_type, isl_num, isl_roi

    def enter_island(self, isl_type, isl_roi):
        """进入指定岛屿"""
        if isl_type == MoonSeaType.island100:
            # 未识别到，随机选择
            self.entry_island_random()
            return
        # 根据OCR坐标计算点击位置
        isl_roi = [isl_roi[0]-40, isl_roi[1]+70, isl_roi[2]+40, isl_roi[3]+40]
        self.C_ISLAND_ENTER.roi_front = isl_roi
        while 1:
            self.screenshot()
            if not self.in_main() and self.appear(self.I_BACK_EXIT): break
            if self.click(self.C_ISLAND_ENTER, interval=2.5): continue

    def activate_store(self) -> bool:
        """激活商店（打Boss前购买技能）"""
        if self.appear(self.I_M_STORE) and not self.appear(self.I_M_STORE_ACTIVITY):
            return False  # 没钱
        while 1:
            self.screenshot()
            if self.appear(self.I_UI_CONFIRM):
                self.ui_click_until_disappear(self.I_UI_CONFIRM)
                break
            if self.appear_then_click(self.I_M_STORE_ACTIVITY, interval=1.5):
                continue
```

### moon_sea/moon_sea.py - 月海主逻辑

```python
class MoonSea(MoonSeaMap, MoonSeaL101, MoonSeaL102, MoonSeaL103, MoonSeaL104, MoonSeaL105):

    def _run_moon_sea(self):
        """月海主循环"""
        limit_time = self._conf.limit_time
        max_cont = self._conf.limit_count
        max_time = timedelta(...)
        cnt = 0
        while 1:
            if cnt >= max_cont: break       # 次数限制
            if datetime.now() - self.start_time >= max_time: break  # 时间限制
            self.one()                       # 执行一次完整流程
            cnt += 1

    def one(self):
        """一次完整的月海流程"""
        self.cnt_skill101 = 1               # 重置柔风计数
        self._start()                        # 开始/继续

        while 1:
            self.screenshot()
            if not self.in_main(): continue

            # 决策：选择下一个岛屿
            isl_type, isl_num, isl_roi = self.decide()

            # 如果是最后一个岛且不是Boss，尝试激活商店
            if isl_num == 1 and isl_type != MoonSeaType.island106:
                if self.cnt_skill101 < 5 and not self.appear(self.I_M_STORE):
                    self.activate_store()
                    # 重新决策
                    isl_type, isl_num, isl_roi = self.decide()
                    if isl_type != MoonSeaType.island101:
                        self.entry_island_random()  # OCR不准时随机进入

            # Boss战斗
            if isl_type == MoonSeaType.island106:
                self.boss_team_lock()
                if self.boss_battle(): break
                else: continue

            # 进入岛屿
            self.enter_island(isl_type, isl_roi)
            isl_type = self.island_name()    # 二次确认岛屿类型
            match isl_type:
                case MoonSeaType.island101: self.run_l101()
                case MoonSeaType.island102: self.run_l102()
                case MoonSeaType.island103: self.run_103()
                case MoonSeaType.island104: self.run_l104()
                case MoonSeaType.island105: self.run_l105()

    def _start(self):
        """开始月海：选择椒图、确认设置、选择首个技能"""
        # 1. 进入六道之门
        while 1:
            if self.appear(self.I_MSTART): break
            if self.appear_then_click(self.I_MENTER, interval=1): continue
            if self.appear(self.I_MCONINUE):
                self._continue()             # 继续上一把
                return

        # 2. 选择椒图
        while 1:
            if self.appear(self.I_MSHOUZU): break
            if self.appear_then_click(self.I_MSHUTEN, interval=3): continue
            if self.appear_then_click(self.I_MSHOUZU_SELECT, interval=1): continue

        # 3. 确认设置并开始
        while 1:
            if self.appear(self.I_PREPARE_BATTLE): break
            if self.appear_then_click(self.I_MSTART_UNCHECK, interval=0.6): continue
            if self.appear_then_click(self.I_UI_CONFIRM, interval=1): continue
            if self.appear_then_click(self.I_MSKIP, interval=1.5): continue
            if self.appear_then_click(self.I_MSTART, interval=3): continue
            if self.appear_then_click(self.I_MSTART_CONFIRM, interval=3): continue

        # 4. 选择首个柔风技能
        while 1:
            if self.appear(self.I_M_STORE): break
            if self.appear_then_click(self.I_MFIRST_SKILL, interval=1): continue

    def island_name(self) -> MoonSeaType:
        """通过OCR识别当前岛屿名称"""
        while 1:
            self.screenshot()
            text = self.O_ISLAND_NAME.ocr(self.device.image)
            if '星' in text: return MoonSeaType.island105
            if '战' in text: return MoonSeaType.island104
            if '混' in text: return MoonSeaType.island103
            if '神秘' in text: return MoonSeaType.island102
            if '宁息' in text: return MoonSeaType.island101

    def boss_battle(self) -> bool:
        """Boss战斗流程"""
        self.ui_click_until_disappear(self.I_BOSS_FIRE, interval=1)
        self.device.stuck_record_add('BATTLE_STATUS_S')
        while 1:
            self.screenshot()
            if self.appear(self.I_BOSS_SHARE): break        # 分享界面=胜利
            if self.appear(self.I_BOSS_BATTLE_GIVEUP):      # 失败
                self.ui_click_until_disappear(self.I_BOSS_BATTLE_GIVEUP)
                continue
            if self.appear(self.I_BOSS_USE_DOUBLE):         # 双倍奖励
                self.ui_get_reward(self.I_BOSS_USE_DOUBLE)
            if self.ui_reward_appear_click(): continue
            if self.appear_then_click(self.I_BOSS_GET_EXP, interval=1): continue
            if self.appear_then_click(self.I_UI_CANCEL, interval=1): continue
            if self.appear_then_click(self.I_BOSS_SKIP, interval=1):  # 第二个boss
                self.device.stuck_record_clear()
                self.device.stuck_record_add('BATTLE_STATUS_S')
                continue
        # 等待结算椒图
        if self.wait_until_appear(self.I_BOSS_SHUTU, wait_time=20):
            self.ui_click(self.I_BOSS_SHUTU, stop=self.I_MSTART)
        return True
```

```python
            if self.in_main(): break
            if self.appear_then_click(self.I_STORE_EXIT, interval=1): continue

    def buy_skill_101(self) -> bool:
        """购买柔风技能"""
        self.wait_animate_stable(self.C_STORE_ANIMATE_KEEP, timeout=2)
        self.wait_until_appear(self.I_STORE_STABLE_FLAG)
        while 1:
            self.screenshot()
            if self.appear_then_click(self.I_UI_CONFIRM_SAMLL, interval=1.5): break
            if self.appear(self.I_UI_CONFIRM): break
            if self.appear(self.I_STORE_SKILL_101, interval=3):
                x, y = self.I_STORE_SKILL_101.front_center()
                x -= 60                    # 偏移点击位置
                self.device.click(x=x, y=y)
        self.ui_click_until_disappear(self.I_UI_CONFIRM)
        self.cnt_skill101 += 1
        return True

    def refresh_store(self) -> bool:
        """刷新商店"""
        text = self.O_STORE_REFRESH_TIME.ocr(self.device.image)
        matches = re.search(r"剩\d+次", text)
        if matches:
            refresh_time = int(matches.group()[1])
            if refresh_time <= 0: return False
        self.appear_then_click(self.I_STORE_REFRESH, interval=1.5)
        self.wait_until_stable(self.I_UI_CONFIRM, timeout=Timer(2, count=10))
        return True
```

```python
                if self.in_main(): break
                if self.appear_then_click(self.I_BACK_EXIT, interval=3): continue
            return
        self.imitate()                                  # 执行仿造

    def imitate(self):
        """仿造技能（复制柔风）"""
        cnt_imitate = 0
        while 1:
            if self.in_main(): break
            if self.appear_then_click(self.I_IMITATE_1, interval=2.5): continue
            if self.appear_then_click(self.I_UI_CONFIRM_SAMLL, interval=1): continue
            if cnt_imitate >= 3: break                  # 最多尝试3次
            if self.appear_then_click(self.I_IMITATE, interval=1):
                cnt_imitate += 1; continue
        self.cnt_skill101 += 1                          # 柔风+1
```

## 6. 设计模式总结

### 1. Roguelike自动化模式
- 通过OCR识别地图上的岛屿类型和剩余回合数
- 根据策略优先选择特定岛屿（宁息买技能 > 战斗 > 神秘）
- 柔风技能为核心：优先购买、仿造、升级到满级

### 2. 深层继承体系
```
BaseTask
  └── MoonSeaSkills (技能/战斗基础)
        ├── MoonSeaMap (地图识别/导航)
        │     └── MoonSea (主逻辑，继承所有岛屿)
        ├── MoonSeaL101 (宁息之岛)
        ├── MoonSeaL102 (神秘之岛)
        ├── MoonSeaL103 (混乱之岛)
        ├── MoonSeaL104 (战斗之岛)
        └── MoonSeaL105 (星之岛)
```

### 3. 自定义OCR管线
- `VerticalText`: 处理竖排文字，通过旋转图片90度转为横排
- `StoneOcr`: 在竖排文字基础上增加石头纹理的图像预处理
- 多层掩码（YUV-U通道、颜色相似度、HSV-V通道）增强文字对比度

### 4. 状态机模式
- `in_main()` 判断是否在月海主界面
- `island_name()` 通过OCR二次确认岛屿类型
- `decide()` 通过地图OCR做出岛屿选择决策

### 5. 策略优先级
- 柔风技能 (cnt_skill101) 为核心目标，满级为5级
- 最后一岛前自动激活商店购买技能
- 技能选择时优先柔风，其次洞察之力
- 金币不足300时退出商店

### 6. 容错与重试
- OCR识别不准时随机选择岛屿
- Boss失败可重试（`boss_battle` 返回 False 继续循环）
- 仿造最多尝试3次
- 商店刷新检查剩余次数

### 7. 动态ROI
- `enter_island()` 根据OCR结果动态设置点击区域的ROI
- `C_ISLAND_ENTER.roi_front = isl_roi` 实现精确点击
- 未识别时使用预定义的6个固定位置 (`C_ISLAND_ENTER_1~6`)

### 8. 动画等待
- `wait_animate_stable()` 等待动画结束
- `wait_until_stable()` 等待UI元素稳定
- 避免在动画过程中误操作
