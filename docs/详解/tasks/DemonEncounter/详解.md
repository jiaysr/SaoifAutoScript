# DemonEncounter 逢魔之时 任务详解

## 1. 文件概述

**任务功能**: 自动执行逢魔之时活动，包括：
- 点灯笼四次
- 处理灯笼事件（宝箱、答题、战斗、结界）
- 挑战逢魔BOSS（普通/极）
- 切换御魂配置
- 答题系统

**入口文件**: `tasks/DemonEncounter/script_task.py`

## 4. 核心代码逐行解释

### ScriptTask.run() - 主入口

```python
def run(self):
    self.conf = self.config.demon_encounter

    # 检查时间（17:00-23:00）
    if not self.check_time():
        raise TaskEnd('DemonEncounter')

    self.ui_get_current_page()

    # 切换御魂
    soul_config = self.config.demon_encounter.demon_soul_config
    best_soul_config = self.config.demon_encounter.best_demon_soul_config
    if soul_config.enable or best_soul_config.enable:
        self.ui_goto(page_shikigami_records)
        self.checkout_soul()

    # 导航到逢魔页面
    self.ui_goto(page_demon_encounter_realworld)

    # 执行灯笼
    self.execute_lantern()

    # 挑战BOSS
    self.execute_boss()

    self.set_next_run(task='DemonEncounter', success=True, finish=False)
    raise TaskEnd('DemonEncounter')
```

### execute_lantern() - 点灯笼

```python
def execute_lantern(self):
    ocr_timer = Timer(0.8)
    ocr_timer.start()

    # 先点四次灯笼
    while 1:
        self.screenshot()
        if not ocr_timer.reached():
            continue
        ocr_timer.reset()

        cu, re, total = self.O_DE_COUNTER.ocr(self.device.image)
        if cu + re != total:
            continue
        if cu == 0 and re == 4:
            break  # 已点4次

        if self.appear_then_click(self.I_DE_FIND, interval=2.5):
            continue

    # 领取红色达摩奖励
    self.screenshot()
    if not self.appear(self.I_DE_AWARD):
        self.ui_get_reward(self.I_DE_RED_DHARMA)
    self.wait_until_appear(self.I_DE_AWARD)

    # 处理四个灯笼事件
    match_click = {
        1: self.C_DE_1, 2: self.C_DE_2,
        3: self.C_DE_3, 4: self.C_DE_4,
    }
    for i in range(1, 5):
        lantern_type = self.check_lantern(i)
        match lantern_type:
            case LanternClass.BOX: self._box(match_click[i])
            case LanternClass.MAIL: self._mail(match_click[i])
            case LanternClass.REALM: self._realm(match_click[i])
            case LanternClass.EMPTY: pass
            case LanternClass.BATTLE: self._battle(match_click[i])
            case LanternClass.MYSTERY: self._mystery(match_click[i])
            case LanternClass.BOSS: self._boss(match_click[i])
```

### check_lantern() - 检查灯笼类型

```python
def check_lantern(self, index: int = 1):
    # 设置ROI
    match_roi = {1: self.C_DE_1.roi_front, ...}
    self.I_DE_BOX.roi_back = match_roi[index]
    self.I_DE_LETTER.roi_back = match_roi[index]
    # ... 其他资源

    self.screenshot()
    if self.appear(self.I_DE_BOX):
        return LanternClass.BOX
    elif self.appear(self.I_DE_LETTER):
        return LanternClass.MAIL
    elif self.appear(self.I_DE_MYSTERY):
        return LanternClass.MYSTERY
    elif self.appear(self.I_DE_REALM):
        return LanternClass.REALM
    elif self.appear(self.I_DE_DEFEAT_1):  # 已击败
        return LanternClass.EMPTY
    elif self.appear(self.I_DE_FIND_BOSS):
        return LanternClass.BOSS
    else:
        return LanternClass.BATTLE  # 默认为战斗
```

### execute_boss() - 挑战BOSS

```python
def execute_boss(self):
    def find_boss():
        timer_find_boss = Timer(10 * 60)  # 10分钟超时
        timer_find_boss.start()
        while 1:
            self.screenshot()
            if self.appear(self.I_BOSS_FIRE) or self.appear(self.I_BEST_BOSS_FIRE):
                break  # 找到BOSS
            if timer_find_boss.reached():
                return False  # 超时

            # 处理50勾玉购买弹窗
            if self.appear(self.I_JADE_50):
                self.ui_click_until_smt_disappear(self.I_DE_FIND, self.I_JADE_50)
                continue

            # 点击寻找BOSS
            if self.best_demon_enable:
                if self.appear(self.I_DE_BOSS_BEST):
                    self.click(self.I_DE_BOSS_BEST)
            else:
                if self.appear(self.I_DE_BOSS):
                    self.click(self.I_DE_BOSS)
        return True

    def enter_boss():
        boss_fire_count = 0
        while 1:
            self.screenshot()
            # 检查人数是否满
            current, remain, total = ocr_people_item.ocr(self.device.image)
            if total == 300 and current >= 290:
                self.ui_click_until_disappear(self.I_UI_BACK_RED)
                return False  # 人满，重新找

            # 确认进入
            if self.appear(self.I_BOSS_CONFIRM):
                self.ui_click(self.I_BOSS_NO_SELECT, self.I_BOSS_SELECTED)
                self.ui_click(self.I_BOSS_CONFIRM, self.I_BOSS_GATHER)
                break
            if self.appear(self.I_BOSS_GATHER):
                break

            if boss_fire_count >= 5:
                return False  # 已挑战过

            if self.appear_then_click(self.I_BOSS_FIRE, interval=3):
                boss_fire_count += 1
        return True

    # 主循环
    fail_count = 0
    while True:
        if fail_count >= 5:
            return
        if not find_boss():
            continue
        if enter_boss():
            break
        fail_count += 1

    # 等待战斗
    time.sleep(5)
    while True:
        self.screenshot()
        if self.appear(self.I_BOSS_DONE_CHECK):
            break  # 战斗结束
        if self.appear(self.I_PREPARE_HIGHLIGHT):
            self.run_general_battle()
```

### _mail() - 答题处理

```python
def _mail(self, target_click):
    def answer():
        self.screenshot()
        question = self.O_LETTER_QUESTION.detect_text(self.device.image)
        answer_1 = self.O_LETTER_ANSWER_1.detect_text(self.device.image)
        answer_2 = self.O_LETTER_ANSWER_2.detect_text(self.device.image)
        answer_3 = self.O_LETTER_ANSWER_3.detect_text(self.device.image)

        # 特殊选项处理
        if answer_1 == '其余选项皆对':
            return 1
        elif answer_2 == '其余选项皆对':
            return 2
        elif answer_3 == '其余选项皆对':
            return 3

        # 使用题库查询
        index = Answer().answer_one(question=question, options=[answer_1, answer_2, answer_3])
        return index if index else 1

    # 打开信件
    while 1:
        self.screenshot()
        if self.appear(self.I_LETTER_CLOSE):
            break
        self.click(target_click)

    # 回答三个问题
    for i in range(1, 4):
        answer_click = answer()
        while 1:
            self.screenshot()
            if self.ui_reward_appear_click():
                break  # 领取奖励
            if not self.appear(self.I_LETTER_CLOSE):
                return  # 答题结束
            self.click(answer_click, interval=1.5)
```

## 6. 设计模式总结

### 1. 策略模式
- `BossType` 按星期选择BOSS
- `best_demon_enable` 选择普通/极模式
- 灯笼类型决定处理方式

### 2. 状态机模式
- `LanternClass` 枚举灯笼状态
- BOSS战斗多阶段状态检测

### 3. 题库系统
- `Answer` 类封装答题逻辑
- OCR识别问题和选项
- 支持特殊选项处理

### 4. 超时控制
- `Timer(10 * 60)` 寻找BOSS超时
- `Timer(5)` 等待战斗超时

### 5. 御魂预设切换
- 按BOSS类型切换不同御魂
- 支持普通/极两套配置

### 6. 时间调度
- `check_time()` 限制活动时间
- 自动推迟到下一个活动时间
