# KekkaiActivation 结界上卡 任务详解

## 1. 文件概述

**任务功能**: 自动执行结界挂卡操作，包括：
- 进入寮结界挂卡界面
- 收取结界卡经验
- 检查卡片状态
- 选择最优卡片挂载
- 处理满级式神

**入口文件**: `tasks/KekkaiActivation/script_task.py`

---

## 2. 配置参数说明

**配置文件**: `tasks/KekkaiActivation/config.py`

### ActivationConfig 参数

| 参数名 | 类型 | 默认值 | 说明 |
|--------|------|--------|------|
| `card_type` | CardType | TAIKO | 挂卡类型（斗鱼/太鼓） |
| `min_taiko_num` | int | 8 | 太鼓每小时最低收益 |
| `min_fish_num` | int | 16 | 斗鱼每小时最低收益 |
| `exchange_before` | bool | True | 挂卡前是否替换满级式神 |
| `exchange_max` | bool | True | 挂卡后是否替换满级式神 |
| `shikigami_class` | ShikigamiClass | N | 式神稀有度 |
| `card_not_found_count` | int | 0 | 未找到卡次数 |

### CardType 枚举

| 值 | 说明 |
|----|------|
| `FISH` | 斗鱼 |
| `TAIKO` | 太鼓 |

### ActivationScheduler

| 参数名 | 类型 | 默认值 | 说明 |
|--------|------|--------|------|
| `priority` | int | 2 | 任务优先级 |
| `success_interval` | TimeDelta | 1天 | 成功后间隔 |
| `failure_interval` | TimeDelta | 10小时 | 失败后间隔 |

## 4. 核心代码逐行解释

### ScriptTask.run() - 主入口

```python
def run(self):
    con = self.config.kekkai_activation.activation_config
    self.ui_get_current_page()
    self.ui_goto(page_guild)

    # 进入寮结界
    self.goto_realm()

    # 挂卡前替换满级式神
    if con.exchange_before:
        self.check_max_lv(con.shikigami_class)

    # 收取经验
    self.harvest_card()

    # 执行挂卡
    self.run_activation(con)

    # 关闭到结界界面
    while 1:
        self.screenshot()
        if self.appear(self.I_REALM_SHIN) or self.appear(self.I_SHI_GROWN):
            break
        if self.appear_then_click(self.I_UI_BACK_RED, interval=1):
            continue

    # 挂卡后替换满级式神
    if con.exchange_max:
        self.check_max_lv(con.shikigami_class)

    self.ui_get_current_page()
    self.ui_goto(page_main)
    raise TaskEnd('KekkaiActivation')
```

### run_activation() - 执行挂卡

```python
def run_activation(self, _config: ActivationConfig) -> bool:
    self.goto_cards()
    logger.hr('Start activation')
    time.sleep(0.5)

    while 1:
        self.screenshot()
        card_status = self.check_card_status()  # 是否有卡
        card_effect = self.check_card_effect()  # 卡是否生效

        # 动画中，等待
        if not card_status and not card_effect:
            if self.appear(self.I_A_ACTIVATE_YELLOW, threshold=0.95):
                continue
            if self.appear(self.I_A_DEMOUNT):
                logger.info('Now in the animation')
                continue

        # 卡正在使用中
        if card_status and card_effect:
            logger.info('Card is using')
            interval = self.ocr_time()
            self.set_next_run("KekkaiActivation", target=interval + datetime.now())
            return False

        # 卡已选中但未激活
        if card_status and not card_effect:
            logger.info('Card is selected but not using')
            while 1:
                self.screenshot()
                if self.appear(self.I_A_INVITE, threshold=0.8):
                    break
                if self.appear_then_click(self.I_UI_CONFIRM, interval=0.6):
                    continue
                if self.appear_then_click(self.I_A_ACTIVATE_YELLOW, interval=1):
                    continue
            interval = self.ocr_time(True)
            self.set_next_run("KekkaiActivation", target=interval + datetime.now())
            return True

        # 无卡状态，开始挂卡
        if not card_status and not card_effect:
            logger.info('Card is not selected also not using')
            self.screening_card(_config.card_type)
```

### screening_card() - 挑选卡片

```python
def screening_card(self, rule: str):
    if rule == CardType.TAIKO:
        card_class = CardClass.TAIKO
        target_class = self.I_A_CARD_KAIKO
    elif rule == CardType.FISH:
        card_class = CardClass.FISH
        target_class = self.I_A_CARD_FISH

    # 打开卡片列表
    while 1:
        self.screenshot()
        if self.appear(target_class):
            time.sleep(0.3)
            self.screenshot()
            if self.appear(target_class):
                break
        if self.click(self.C_A_SELECT_CARD_LIST, interval=2.5):
            continue

    # 选择卡片类型
    while 1:
        self.screenshot()
        if not self.appear(target_class):
            break
        if self.appear_then_click(target_class, interval=1):
            continue

    # 找最优卡
    while 1:
        self.screenshot()
        target = self.check_card_num()
        if target is None:
            self._card_not_found()
        if self.appear(self.I_A_EMPTY):
            while 1:
                self.screenshot()
                if not self.appear(self.I_A_EMPTY):
                    return  # 挂卡成功
                if self.click(target, interval=1):
                    continue
```

### check_card_num() - 检查卡片数值

```python
def check_card_num(self):
    rule = self.config.kekkai_activation.activation_config.card_type
    if rule == CardType.TAIKO:
        min_card_num = self.config.kekkai_activation.activation_config.min_taiko_num
        check_card = "勾玉"
    elif rule == CardType.FISH:
        min_card_num = self.config.kekkai_activation.activation_config.min_fish_num
        check_card = "体力"

    while 1:
        self.screenshot()
        results = self.O_CHECK_CARD_NUMBER.detect_and_ocr(self.device.image)

        # 筛选包含目标文字的结果
        filtered_results = [r for r in results if check_card in r.ocr_text]

        # 提取数字并排序
        numeric_results = []
        for result in filtered_results:
            numbers = [int(num) for num in re.findall(r'\d+', result.ocr_text)]
            if numbers and numbers[0] >= min_card_num:
                numeric_results.append((numbers[0], result))

        if numeric_results:
            # 选择数值最大的
            sorted_results = sorted(numeric_results, key=lambda x: x[0], reverse=True)
            max_result = sorted_results[0][1]
            # 计算点击位置
            box = max_result.box
            x_min = self.O_CHECK_CARD_NUMBER.roi[0] + box[0][0]
            y_min = self.O_CHECK_CARD_NUMBER.roi[1] + box[0][1]
            width = box[1][0] - box[0][0]
            height = box[2][1] - box[1][1]
            roi = int(x_min), int(y_min), int(width), int(height)
            return RuleClick(roi_front=roi, roi_back=roi, name="tmpclick")
        else:
            # 滑动继续查找
            self.device.swipe_adb(p1, p2, duration=2)
            time.sleep(1)
```

### _card_not_found() - 未找到卡处理

```python
def _card_not_found(self):
    activation_config = self.config.kekkai_activation.activation_config
    retry_minutes = 180
    retry_count = 3

    activation_config.card_not_found_count += 1

    if activation_config.card_not_found_count >= retry_count:
        # 达到重试上限，延长下次执行
        log_msg = f"⚠️{activation_config.card_type}卡未检出，{retry_minutes}分钟后重试"
        activation_config.card_not_found_count = 0
        next_run = datetime.now() + timedelta(minutes=retry_minutes)
    else:
        # 切换卡类型
        new_type = CardType.FISH if activation_config.card_type == CardType.TAIKO else CardType.TAIKO
        log_msg = f"🔄{activation_config.card_type}卡未检出 → 切换{new_type}"
        activation_config.card_type = new_type
        next_run = datetime.now()

    self.save_image(content=log_msg, push_flag=True)
    self.config.save()
    self.set_next_run("KekkaiActivation", target=next_run)
    raise TaskEnd
```

## 6. 设计模式总结

### 1. 状态机模式
- 4种卡片状态：无卡、有卡未激活、有卡已激活、动画中
- 根据状态执行不同操作

### 2. 智能选卡算法
- OCR识别卡片数值
- 按数值排序选择最优
- 滑动遍历查找

### 3. 重试机制
- `card_not_found_count` 计数
- 达到上限切换卡类型
- 3次未找到延长执行间隔

### 4. 继承复用
- 继承 `KekkaiUtilize` 的 `ScriptTask`
- 复用寮结界导航、式神替换等功能

### 5. 定时任务模式
- 根据卡片剩余时间设置下次执行
- 失败时3小时后重试
