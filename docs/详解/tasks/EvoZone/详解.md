# 觉醒副本任务 (EvoZone) 逐行代码详解

## 1. 文件概述

觉醒副本任务用于自动刷觉醒材料副本（火/风/水/雷麒麟）。该任务支持四种运行模式：

- **队长模式 (LEADER)**：创建队伍、邀请队友、循环挑战
- **队员模式 (MEMBER)**：接受邀请、等待队长开始、跟随战斗
- **单人模式 (ALONE)**：独自挑战觉醒副本
- **野队模式 (WILD)**：匹配野队（未实现）

### 觉醒副本类型

| 类型 | 说明 |
|------|------|
| 火麒麟 (FIREKIRIN) | 火属性觉醒材料 |
| 风麒麟 (WINDKIRIN) | 风属性觉醒材料 |
| 水麒麟 (WATERKIRIN) | 水属性觉醒材料 |
| 雷麒麟 (LIGHTNINGKIRIN) | 雷属性觉醒材料 |

### 文件结构

| 文件 | 说明 |
|------|------|
| `script_task.py` | 任务入口，包含主流程和各模式实现 |
| `config.py` | 配置定义，包含麒麟类型、层数、限制等 |
| `assets.py` | 资源定义，包含图像匹配、列表、OCR等规则 |

## 4. 核心代码逐行解释

### 4.1 `script_task.py` - 主流程

#### ScriptTask 类定义
```python
class ScriptTask(GeneralBattle, GeneralInvite, GeneralBuff, GeneralRoom, 
                 GameUi, EvoZoneAssets, SwitchSoul):
```
- **多重继承**：集成了战斗、邀请、加成、房间、UI、资源和御魂切换功能

#### run() - 主入口
```python
def run(self) -> bool:
    # 1. 初始化限制参数
    limit_count = self.config.evo_zone.evo_zone_config.limit_count
    limit_time = self.config.evo_zone.evo_zone_config.limit_time
    self.current_count = 0
    self.limit_count: int = limit_count
    self.limit_time: timedelta = timedelta(hours=limit_time.hour, 
                                            minutes=limit_time.minute,
                                            seconds=limit_time.second)

    # 2. 切换御魂
    con = self.config.evo_zone
    if con.switch_soul_config.enable:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul(con.switch_soul_config.switch_group_team)
    if con.switch_soul_config.enable_switch_by_name:
        self.ui_get_current_page()
        self.ui_goto(page_shikigami_records)
        self.run_switch_soul_by_name(con.switch_soul_config.group_name, 
                                      con.switch_soul_config.team_name)

    # 3. 开启觉醒加成（如果配置）
    self.ui_get_current_page()
    self.ui_goto(page_main)
    config: EvoZone = self.config.evo_zone
    if config.evo_zone_config.soul_buff_enable:
        self.open_buff()
        self.awake(is_open=True)
        self.close_buff()

    # 4. 根据身份执行
    success = True
    match config.evo_zone_config.user_status:
        case UserStatus.LEADER:
            success = self.run_leader()
        case UserStatus.MEMBER:
            success = self.run_member()
        case UserStatus.ALONE:
            self.run_alone()
        case UserStatus.WILD:
            self.run_wild()  # 未实现

    # 5. 关闭觉醒加成
    if config.evo_zone_config.soul_buff_enable:
        self.open_buff()
        self.awake(is_open=False)
        self.close_buff()

    # 6. 设置下次运行时间
    if success:
        self.set_next_run('EvoZone', finish=True, success=True)
    else:
        self.set_next_run('EvoZone', finish=False, success=False)

    raise TaskEnd
```

#### evozone_enter() - 进入觉醒副本
```python
def evozone_enter(self) -> bool:
    """根据配置的麒麟类型进入对应的觉醒副本"""
    kirintype = self.I_LIGHTNING_KIRIN  # 默认雷麒麟
    match self.config.evo_zone.evo_zone_config.kirin_type:
        case KirinType.FIREKIRIN: kirintype = self.I_FIRE_KIRIN
        case KirinType.WINDKIRIN: kirintype = self.I_WIND_KIRIN
        case KirinType.WATERKIRIN: kirintype = self.I_WATER_KIRIN
        case KirinType.LIGHTNINGKIRIN: kirintype = self.I_LIGHTNING_KIRIN

    while True:
        self.screenshot()
        if self.appear(self.I_FORM_TEAM):
            return True  # 成功进入
        if self.appear_then_click(kirintype, interval=1):
            continue  # 点击麒麟入口
```

#### check_layer() - 检查层数

def check_layer(self, layer: str) -> bool:
    """检查并选择挑战层数"""
    pos = self.list_find(self.L_LAYER_LIST, layer)
    if pos:
        self.device.click(x=pos[0], y=pos[1])
        return True

#### check_lock() - 检查锁定状态
```python
def check_lock(self, lock: bool = True) -> bool:
    """检查并设置阵容锁定状态"""
    if lock:
        while 1:
            self.screenshot()
            if self.appear(self.I_EVOZONE_LOCK):
                return True  # 已锁定
            if self.appear_then_click(self.I_EVOZONE_UNLOCK, interval=1):
                continue  # 点击锁定
    else:
        while 1:
            self.screenshot()
            if self.appear(self.I_EVOZONE_UNLOCK):
                return True  # 已解锁
            if self.appear_then_click(self.I_EVOZONE_LOCK, interval=1):
                continue  # 点击解锁
```

#### run_leader() - 队长模式
```python
def run_leader(self):
    logger.info('Start run leader')

    # 1. 导航到觉醒界面
    self.ui_get_current_page()
    self.ui_goto(page_awake_zones)
    self.evozone_enter()

    # 2. 选择层数和锁定状态
    layer = self.config.evo_zone.evo_zone_config.layer
    self.check_layer(layer)
    self.check_lock(self.config.evo_zone.general_battle_config.lock_team_enable)

    # 3. 创建队伍
    while 1:
        self.screenshot()
        if self.appear(self.I_CHECK_TEAM):
            break
        if self.appear_then_click(self.I_FORM_TEAM, interval=1):
            continue

    # 4. 创建房间
    self.create_room()
    self.ensure_private()
    self.create_ensure()

    # 5. 邀请队友循环
    success = True
    is_first = True
    while 1:
        self.screenshot()

        # 处理邀请确认弹窗
        if self.check_and_invite(self.config.evo_zone.invite_config.default_invite):
            continue

        # 检查猫咪奖励
        if self.appear_then_click(self.I_PET_PRESENT, action=self.C_WIN_3, interval=1):
            continue

        # 检查次数限制
        if self.current_count >= self.limit_count:
            if self.is_in_room():
                break

        # 检查时间限制
        if datetime.now() - self.start_time >= self.limit_time:
            if self.is_in_room():
                break

        # 检查是否还在房间
        if not self.is_in_room():
            sleep(0.5)
            if self.appear(self.I_MATCHING) or self.appear(self.I_CHECK_EXPLORATION):
                sleep(0.5)
                if self.appear(self.I_MATCHING) or self.appear(self.I_CHECK_EXPLORATION):
                    success = False  # 房间已解散
                    break
            continue

        # 邀请队友并战斗
        if not is_first:
            if self.run_invite(config=self.config.evo_zone.invite_config):
                self.run_general_battle(config=self.config.evo_zone.general_battle_config)
            else:
                success = False  # 邀请失败
                break

        # 第一次邀请
        if is_first:
            if not self.run_invite(config=self.config.evo_zone.invite_config, is_first=True):
                success = False
                break
            else:
                is_first = False
                self.run_general_battle(config=self.config.evo_zone.general_battle_config)

    # 6. 退出房间
    if self.exit_room(): pass
    if self.exit_team(): pass

    self.ui_get_current_page()
    self.ui_goto(page_main)
    return success
```

#### run_member() - 队员模式
```python
def run_member(self):
    logger.info('Start run member')

    # 进入战斗流程
    self.device.stuck_record_add('BATTLE_STATUS_S')

    while 1:
        self.screenshot()

        # 检查猫咪奖励
        if self.appear_then_click(self.I_PET_PRESENT, action=self.C_WIN_3, interval=1):
            continue

        # 检查次数限制
        if self.current_count >= self.limit_count:
            break

        # 检查时间限制
        if datetime.now() - self.start_time >= self.limit_time:
            break

        # 接受邀请
        if self.check_then_accept():
            continue

        # 在房间中等待
        if self.is_in_room():
            self.device.stuck_record_clear()
            if self.wait_battle(wait_time=self.config.evo_zone.invite_config.wait_time):
                self.run_general_battle(config=self.config.evo_zone.general_battle_config)
            else:
                break  # 等待超时
        # 队长秒开的情况
        elif self.check_take_over_battle(False, config=self.config.evo_zone.general_battle_config):
            continue

    # 退出循环
    while 1:
        if self.appear(self.I_GI_HOME) or self.appear(self.I_GI_EXPLORE):
            break
        if self.exit_room(): pass
        if self.exit_battle(): pass

    self.ui_get_current_page()
    self.ui_goto(page_main)
    return True
```

#### run_alone() - 单人模式
```python
def run_alone(self):
    logger.info('Start run alone')

    # 1. 导航到觉醒界面
    self.ui_get_current_page()
    self.ui_goto(page_awake_zones)
    self.evozone_enter()

    # 2. 选择层数和锁定状态
    layer = self.config.evo_zone.evo_zone_config.layer
    self.check_layer(layer)
    self.check_lock(self.config.evo_zone.general_battle_config.lock_team_enable)

    # 3. 定义辅助函数
    def is_in_evozone(screenshot=False) -> bool:
        if screenshot: self.screenshot()
        return self.appear(self.I_EVOZONE_FIRE)

    # 4. 挑战循环
    while 1:
        self.screenshot()

        # 检查猫咪奖励
        if self.appear_then_click(self.I_PET_PRESENT, action=self.C_WIN_3, interval=1):
            continue

        if not is_in_evozone(): continue

        # 检查次数限制
        if self.current_count >= self.limit_count:
            break

        # 检查时间限制
        if datetime.now() - self.start_time >= self.limit_time:
            break

        # 点击挑战
        while 1:
            self.screenshot()
            if self.appear_then_click(self.I_EVOZONE_FIRE, interval=1):
                pass
            if not self.appear(self.I_EVOZONE_FIRE):
                self.run_general_battle(config=self.config.evo_zone.general_battle_config)
                break

    # 5. 返回
    while 1:
        self.screenshot()
        if not self.appear(self.I_FORM_TEAM): break
        if self.appear_then_click(self.I_BACK_Y, interval=1): continue

    self.ui_current = page_awake_zones
    self.ui_goto(page_main)
```

## 6. 设计模式总结

### 6.1 策略模式
- `UserStatus` 定义不同身份
- `run_leader()` / `run_member()` / `run_alone()` 封装不同策略
- 通过 `match` 语句动态选择

### 6.2 组合模式
- 多重继承组合功能模块：
  - `GeneralBattle` - 通用战斗
  - `GeneralInvite` - 邀请系统
  - `GeneralBuff` - 加成管理
  - `GeneralRoom` - 房间管理
  - `GameUi` - UI导航
  - `EvoZoneAssets` - 资源定义
  - `SwitchSoul` - 御魂切换

### 6.3 模板方法模式
- `run()` 定义任务骨架
- 子方法实现具体步骤
- 前置处理（御魂切换、加成开启）和后置处理（加成关闭）统一管理

### 6.4 资源驱动模式
- `RuleImage` 定义图像匹配规则
- `RuleList` 定义列表选择规则
- `RuleOcr` 定义文字识别规则
- 代码通过资源引用操作UI

### 6.5 配置驱动模式
- 所有参数通过 `config.py` 定义
- 支持运行时动态调整
- 使用 Pydantic 进行数据验证

### 6.6 容错机制
- **超时检测**：防止无限等待
- **次数限制**：防止过度消耗
- **房间检测**：防止房间解散后卡住
- **猫咪奖励**：自动领取宠物发现的宝藏
- **状态恢复**：退出时确保回到主页

### 6.7 加成管理
- 进入任务前开启觉醒加成
- 退出任务后关闭觉醒加成
- 使用 `GeneralBuff` 统一管理
